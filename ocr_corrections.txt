# OCR Corrections — Known misreads and their fixes
# Format: wrong_text -> correct_text
# Lines starting with # are comments. Blank lines are ignored.
# Phrases (multi-word) are also supported.

## Explicit Word Corrections
# Germination misreads — VLM confuses handwritten G with D/H/J
cermination -> germination
gernination -> germination
germiation -> germination
gerrn -> germ
gern -> germ
herm -> germ
jerm -> germ
jerns -> germ
# Note: dorm. -> germ. is handled as a contextual fix (not token-level)
# because "dorm" is a valid abbreviation for dormancy in curator's notes
# Sown misreads — S read as D or digits
down -> sown
soon -> sown
# Note: 50wn and s0wn are handled as contextual fixes (contain digits)
# Seedling misreads
suddling -> seedling
suddlings -> seedlings
seediing -> seedling
seediings -> seedlings
# Other action misreads
prickled -> pricked
pickled -> pricked
dull -> full
# Material/chemical misreads
osmosote -> osmocote
osmocotee -> osmocote
nutricole -> nutricote
periite -> perlite
vermicuiite -> vermiculite
coldfrarne -> coldframe
coldfrarne. -> coldframe
glasthouse -> glasshouse
shadehause -> shadehouse
# Other
buibs -> bulbs
offsefs -> offsets
scionss -> scions

## Phrase Corrections
prickled out -> pricked out
picked out -> pricked out
surfacesow -> surface sow
